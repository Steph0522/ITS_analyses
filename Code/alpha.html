get_letters <- function(data, y_pos) {
  m <- aov(lm(val ~ Poligono * Season, data = data))
  p <- summary(m)[[1]][["Pr(>F)"]][1:3]

  if (p[3] < 0.05) {   # interacción significativa
    lets <- HSD.test(aov(lm(val ~ interact, data = data)), "interact")$groups %>%
      rownames_to_column("interact") %>%
      separate(interact, c("Poligono", "Season"), remove = FALSE) %>%
      mutate(xpos = as.numeric(factor(Season, levels = c("Dry", "Rainy"))))
  } else {             # solo polígono
    lets <- HSD.test(aov(lm(val ~ Poligono, data = data)), "Poligono")$groups %>%
      rownames_to_column("Poligono") %>%
      mutate(xpos = 1.5)
  }
  mutate(lets, val = y_pos)
}
